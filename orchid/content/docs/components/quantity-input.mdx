## Usage

```tsx
import { useState } from 'react'
import { QuantityInput } from '@/components/quantity-input'

function QuantityExample() {
  const [quantity, setQuantity] = useState(2)

  return (
    <QuantityInput
      value={quantity}
      min={1}
      max={99}
      onValueChange={setQuantity}
    />
  )
}
```
